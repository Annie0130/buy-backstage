// 上架控制台：在自己電腦開一個網頁，用按鈕操作所有上架功能
// 用法：雙擊「控制台.bat」（或 node panel.js），瀏覽器會自動打開 http://localhost:3939
// 只有這台電腦自己連得到（127.0.0.1），別人連不進來
const http = require('http');
const fs = require('fs');
const path = require('path');
const { spawn, exec } = require('child_process');

// 舊的資料夾排法 → 先整理資料夾（最外層只留按鈕和照片資料夾），再從「系統」裡開控制台
if (fs.existsSync(path.join(__dirname, 'migrate.js'))) {
  const M = require('./migrate.js');
  if (M.needsMigration(__dirname) && M.migrate(__dirname)) { process.chdir(path.join(__dirname, M.SYS)); require(path.join(__dirname, M.SYS, 'panel.js')); return; }
}
const PORT = 3939;
// 用電腦預設的程式打開（Windows：start；Mac：open）
function openPath(p) {
  if (process.platform === 'win32') exec(`start "" "${p}"`);
  else if (process.platform === 'darwin') exec(`open "${p.replace(/"/g, '\\"')}"`);
}
const DIR = __dirname;
const ROOT = path.basename(DIR) === '系統' ? path.dirname(DIR) : DIR;
const INBOX = ROOT !== DIR ? path.join(ROOT, '照片放這裡') : path.join(DIR, '商品照片', '收件匣');
const REFILL_FILE = path.join(DIR, '補照片清單.txt');
const PREVIEW = path.join(DIR, '照片分組預覽.html');

// 只允許這些指令（不能從網頁執行任意東西）
const COMMANDS = {
  order: { label: '📷 拍照順序', args: ['upload.js', 'photos', '--order'] },
  all: { label: '🚀 一鍵上架', args: ['upload.js', 'all'] },
  group: { label: '🗂️ 只分組照片', args: ['upload.js', 'photos'] },
  addphotos: { label: '🖼️ 補照片', args: ['upload.js', 'addphotos'], serials: true },
  draft: { label: '📝 全部存草稿', args: ['upload.js', 'airtable', '--draft'] },
  drafts: { label: '📝 草稿補照片並發佈', args: ['upload.js', 'drafts'] },
  shopee: { label: '🦐 蝦皮補資料', args: ['upload.js', 'shopee', '--auto'] },
  check: { label: '💰 蝦皮價格檢查', args: ['upload.js', 'shopee', '--auto'] },          // 只處理還沒補好的
  recheck: { label: '💰 全部重新檢查蝦皮', args: ['upload.js', 'shopee', '--auto', '--check'] },
  bsmi: { label: '🧾 批次補 BSMI', args: ['upload.js', 'bsmifix'] },
  catfix: { label: '🗂️ 批次改蝦皮分類', args: ['upload.js', 'catfix'] },
  republish: { label: '🔁 重新發佈到蝦皮', args: ['upload.js', 'republish'], serials: true },
  todo: { label: '📋 待辦清單', args: ['upload.js', 'todo'] },
  markshopee: { label: '✓ 蝦皮已手動補好', args: ['upload.js', 'markshopee'], serials: true },
  update: { label: '🔄 更新程式', args: ['update.js'] },
};

let job = null;          // { key, label, child, start }
let logBuf = [];         // 最近的輸出
const clients = new Set();
function broadcast(type, data) {
  const msg = `event: ${type}\ndata: ${JSON.stringify(data)}\n\n`;
  for (const res of clients) res.write(msg);
}
function pushLog(text) {
  logBuf.push(text);
  if (logBuf.length > 4000) logBuf = logBuf.slice(-3000);
  broadcast('log', text);
}
const jobInfo = () => (job ? { running: true, key: job.key, label: job.label, start: job.start } : { running: false });

function runJob(key, serials = []) {
  if (job) return { ok: false, msg: `「${job.label}」還在跑，請等它跑完或按停止` };
  const c = COMMANDS[key];
  if (!c) return { ok: false, msg: '沒有這個功能' };
  const extra = c.serials ? serials.filter((s) => /^[\w-]{1,30}$/.test(s)) : [];
  logBuf = [];
  const child = spawn(process.execPath, [...c.args, ...extra], { cwd: DIR, env: { ...process.env, FORCE_COLOR: '0', PANEL: '1' }, detached: process.platform !== 'win32' });
  job = { key, label: c.label + (extra.length ? ' ' + extra.join(' ') : ''), child, start: Date.now() };
  pushLog(`▶ 開始：${job.label}\n\n`);
  broadcast('job', jobInfo());
  child.stdout.setEncoding('utf8');
  child.stderr.setEncoding('utf8');
  child.stdout.on('data', pushLog);
  child.stderr.on('data', pushLog);
  child.on('close', (code) => {
    pushLog(`\n■ 結束${code ? `（代碼 ${code}）` : ''}\n`);
    job = null;
    broadcast('job', jobInfo());
    broadcast('refresh', {});
  });
  return { ok: true };
}
function stopJob() {
  if (!job) return;
  const pid = job.child.pid;
  // Windows 要連同它開的 Chrome 一起關
  if (process.platform === 'win32') exec(`taskkill /pid ${pid} /T /F`);
  else { try { process.kill(-pid, 'SIGTERM'); } catch (e) { job.child.kill('SIGTERM'); } } // Mac：連同它開的 Chrome 一起關
  pushLog('\n⏹ 已按停止\n');
}

function getStatus() {
  return new Promise((resolve) => {
    const child = spawn(process.execPath, ['upload.js', 'status'], { cwd: DIR });
    let out = '';
    child.stdout.setEncoding('utf8');
    child.stdout.on('data', (d) => (out += d));
    child.stderr.on('data', (d) => (out += d));
    const timer = setTimeout(() => child.kill(), 90000);
    child.on('close', () => {
      clearTimeout(timer);
      const line = out.split('\n').find((l) => l.startsWith('__STATUS__'));
      if (line) { try { return resolve({ ok: true, ...JSON.parse(line.slice(10)) }); } catch (e) {} }
      resolve({ ok: false, msg: out.trim().split('\n').slice(-3).join('\n') || '讀不到狀態' });
    });
  });
}

const REFILL_HEADER = '# 補照片清單：已經上架、要補拍照片的商品，照拍照順序一行打一個序號\r\n# （# 開頭的行不會被讀取；清單是空的就照收購後台的標記）\r\n';
function readRefill() {
  if (!fs.existsSync(REFILL_FILE)) return [];
  return fs.readFileSync(REFILL_FILE, 'utf8').replace(/^﻿/, '').split(/\r?\n/).map((l) => l.trim()).filter((l) => l && !l.startsWith('#'));
}
function writeRefill(list) {
  const clean = list.map((s) => String(s).trim()).filter((s) => /^[\w-]{1,30}$/.test(s));
  fs.writeFileSync(REFILL_FILE, '﻿' + REFILL_HEADER + clean.join('\r\n') + (clean.length ? '\r\n' : ''), 'utf8');
  return clean;
}

function body(req) {
  return new Promise((resolve) => {
    let b = '';
    req.on('data', (d) => { b += d; if (b.length > 1e5) req.destroy(); });
    req.on('end', () => { try { resolve(JSON.parse(b || '{}')); } catch (e) { resolve({}); } });
  });
}
const json = (res, obj) => { res.writeHead(200, { 'Content-Type': 'application/json; charset=utf-8' }); res.end(JSON.stringify(obj)); };

const server = http.createServer(async (req, res) => {
  const url = new URL(req.url, 'http://localhost');
  // 只接受從本機網頁送來的操作
  if (req.method === 'POST') {
    const origin = req.headers.origin || '';
    if (origin && !/^http:\/\/(localhost|127\.0\.0\.1):3939$/.test(origin)) { res.writeHead(403); return res.end(); }
  }
  try {
    if (url.pathname === '/') {
      res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' });
      return res.end(fs.readFileSync(path.join(DIR, 'panel.html')));
    }
    if (url.pathname === '/api/events') {
      res.writeHead(200, { 'Content-Type': 'text/event-stream', 'Cache-Control': 'no-cache', Connection: 'keep-alive' });
      res.write(`event: job\ndata: ${JSON.stringify(jobInfo())}\n\n`);
      res.write(`event: history\ndata: ${JSON.stringify(logBuf.join(''))}\n\n`);
      clients.add(res);
      req.on('close', () => clients.delete(res));
      return;
    }
    if (url.pathname === '/api/status') return json(res, await getStatus());
    if (url.pathname === '/api/commands') return json(res, Object.fromEntries(Object.entries(COMMANDS).map(([k, v]) => [k, v.label])));
    if (url.pathname === '/api/run' && req.method === 'POST') { const b = await body(req); return json(res, runJob(b.cmd, b.serials || [])); }
    if (url.pathname === '/api/input' && req.method === 'POST') {
      const b = await body(req);
      if (!job) return json(res, { ok: false, msg: '目前沒有在跑' });
      const text = String(b.text || '').slice(0, 200);
      job.child.stdin.write(text + '\n');
      pushLog(`${text || '（Enter）'}\n`);
      return json(res, { ok: true });
    }
    if (url.pathname === '/api/stop' && req.method === 'POST') { stopJob(); return json(res, { ok: true }); }
    if (url.pathname === '/api/refill') {
      if (req.method === 'POST') { const b = await body(req); return json(res, { ok: true, list: writeRefill(b.list || []) }); }
      return json(res, { ok: true, list: readRefill() });
    }
    if (url.pathname === '/api/ncc' && req.method === 'POST') {
      // 在控制台補 NCC／BSMI：寫進 NCC字號.csv（同型號就更新，並清掉 ⚠️ 備註）
      const b = await body(req);
      const model = String(b.model || '').trim(), ncc = String(b.ncc || '').trim(), bsmi = String(b.bsmi || '').trim();
      if (!model || model.length > 60 || /[,"\r\n]/.test(model + ncc + bsmi)) return json(res, { ok: false, msg: '型號不能空白，也不能有逗號' });
      if (!ncc && !bsmi) return json(res, { ok: false, msg: 'NCC 和 BSMI 至少填一個' });
      const f = path.join(DIR, 'NCC字號.csv');
      const nk = (t) => String(t || '').toLowerCase().replace(/[\s\-_]/g, '');
      const lines = fs.existsSync(f) ? fs.readFileSync(f, 'utf8').replace(/^\uFEFF/, '').split(/\r?\n/).filter(Boolean) : ['型號,NCC字號,BSMI字號,備註'];
      let hit = false;
      const out = lines.map((l, i) => {
        if (i === 0) return l;
        const c = l.split(',');
        if (nk(c[0]) !== nk(model)) return l;
        hit = true;
        return [c[0], ncc || c[1] || '', bsmi || c[2] || '', '在控制台補的'].join(',');
      });
      if (!hit) out.push([model, ncc, bsmi, '在控制台補的'].join(','));
      fs.writeFileSync(f, '\uFEFF' + out.join('\r\n') + '\r\n', 'utf8');
      return json(res, { ok: true });
    }
    if (url.pathname === '/api/open' && req.method === 'POST') {
      // 打開資料夾（只限這幾個）
      const b = await body(req);
      const target = { inbox: INBOX, photos: path.join(DIR, '商品照片'), root: ROOT, bsmi: path.join(DIR, 'BSMI補填.csv') }[b.what];
      if (target) { if (b.what !== 'bsmi') fs.mkdirSync(target, { recursive: true }); if (fs.existsSync(target)) openPath(target); }
      return json(res, { ok: !!target });
    }
    if (url.pathname === '/preview') {
      if (!fs.existsSync(PREVIEW)) { res.writeHead(404, { 'Content-Type': 'text/plain; charset=utf-8' }); return res.end('還沒有分組預覽'); }
      res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' });
      return res.end(fs.readFileSync(PREVIEW));
    }
    if (url.pathname === '/api/preview-time') return json(res, { t: fs.existsSync(PREVIEW) ? fs.statSync(PREVIEW).mtimeMs : 0 });
    res.writeHead(404); res.end();
  } catch (e) {
    res.writeHead(500, { 'Content-Type': 'text/plain; charset=utf-8' });
    res.end('錯誤：' + e.message);
  }
});

server.on('error', (e) => {
  if (e.code === 'EADDRINUSE') {
    console.log('控制台已經開著了，直接打開瀏覽器…');
    openPath(`http://localhost:${PORT}`);
    setTimeout(() => process.exit(0), 1500);
  } else { console.error(e); process.exit(1); }
});
server.listen(PORT, '127.0.0.1', () => {
  console.log(`上架控制台已開啟：http://localhost:${PORT}`);
  console.log('（這個黑色視窗不要關，關掉控制台就停了）');
  if (!process.argv.includes('--no-open')) openPath(`http://localhost:${PORT}`);
});
