<script>
const $ = (id) => document.getElementById(id);
const esc = (s) => String(s ?? '').replace(/[&<>"]/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]));
function toast(msg) { const t = $('toast'); t.textContent = msg; t.classList.add('show'); clearTimeout(t._t); t._t = setTimeout(() => t.classList.remove('show'), 2500); }
async function post(url, data) { const r = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data || {}) }); return r.json(); }

// ---------- 執行 ----------
let running = false, tail = '', dangers = [];
async function run(cmd, serials) {
  if (running) return toast('還有功能在跑，請等它跑完');
  const r = await post('/api/run', { cmd, serials });
  if (!r.ok) toast(r.msg);
}
function runAddPhotos() {
  const s = $('serials').value.trim().split(/[\s,，、]+/).filter(Boolean);
  if (!s.length && !confirm('沒有輸入序號，要自動找「所有有新照片的商品」嗎？')) return;
  run('addphotos', s);
}
async function send(text) { await post('/api/input', { text }); $('replyText').value = ''; $('reply').classList.remove('show'); tail = ''; }
async function stop() { if (confirm('確定要停止嗎？（跑到一半的那一件可能要到飛鼠手動確認）')) await post('/api/stop'); }
function openFolder(what) { post('/api/open', { what }); }
function openPreview() { window.open('/preview?t=' + Date.now(), '_blank'); }

function lineClass(l) {
  if (/🚨|❌|失敗|錯誤/.test(l)) return 'bad';
  if (/⚠️|警告/.test(l)) return 'warn';
  if (/✅|完成|成功/.test(l)) return 'ok';
  return '';
}
function appendLog(text) {
  const log = $('log');
  const atBottom = log.scrollTop + log.clientHeight >= log.scrollHeight - 30;
  const parts = text.split('\n');
  parts.forEach((l, i) => {
    const cls = lineClass(l);
    log.insertAdjacentHTML('beforeend', (cls ? `<span class="${cls}">${esc(l)}</span>` : esc(l)) + (i < parts.length - 1 ? '\n' : ''));
    if (l.includes('🚨') && !dangers.includes(l.trim())) { dangers.push(l.trim()); showAlert(); }
  });
  if (atBottom) log.scrollTop = log.scrollHeight;
  // 偵測「在等你回覆」：最後一段沒有換行、像是問題
  tail = (tail + text).slice(-300);
  const last = tail.split('\n').pop();
  if (running && last && /[：:?？]\s*$/.test(last)) {
    $('replyQ').textContent = '程式在等你回覆：' + last.trim();
    $('reply').classList.add('show');
  } else if (!last) $('reply').classList.remove('show');
}
function showAlert() {
  const a = $('alert');
  a.innerHTML = '🚨 有商品要馬上處理（蝦皮價格／最低購買數量）：<br>' + dangers.map(esc).join('<br>');
  a.classList.add('show');
}
let timer = null;
function setJob(j) {
  running = j.running;
  $('stopBtn').disabled = !j.running;
  document.querySelectorAll('main button:not(#stopBtn):not(.reply button)').forEach((b) => { if (!b.closest('.reply') && !b.closest('.filters') && !b.closest('.chips')) b.disabled = j.running && !/重新整理|看分組預覽|打開/.test(b.textContent); });
  clearInterval(timer);
  if (j.running) {
    const tick = () => { const s = Math.round((Date.now() - j.start) / 1000); $('pill').textContent = `執行中：${j.label}　${Math.floor(s / 60)}:${String(s % 60).padStart(2, '0')}`; };
    tick(); timer = setInterval(tick, 1000);
    $('pill').className = 'pill run';
    $('jobInfo').textContent = j.label;
  } else {
    $('pill').textContent = '閒置中'; $('pill').className = 'pill';
    $('reply').classList.remove('show');
  }
}
const es = new EventSource('/api/events');
es.addEventListener('job', (e) => { const j = JSON.parse(e.data); if (j.running && !running) { $('log').innerHTML = ''; dangers = []; $('alert').classList.remove('show'); } setJob(j); });
es.addEventListener('history', (e) => { $('log').innerHTML = ''; appendLog(JSON.parse(e.data)); });
es.addEventListener('log', (e) => appendLog(JSON.parse(e.data)));
es.addEventListener('refresh', () => { loadStatus(); loadRefill(); });
es.onerror = () => { $('pill').textContent = '⚠️ 控制台沒有在跑（黑色視窗被關掉了？）'; $('pill').className = 'pill run'; };

// ---------- 補照片清單 ----------
let refill = [];
async function loadRefill() { const r = await (await fetch('/api/refill')).json(); refill = r.list || []; drawChips(); drawTable(); }
async function saveRefill(list) { const r = await post('/api/refill', { list }); refill = r.list; drawChips(); drawTable(); }
function addRefill(sn) {
  const add = (sn || $('refillInput').value).trim().split(/[\s,，、]+/).filter(Boolean);
  $('refillInput').value = '';
  saveRefill([...refill, ...add.filter((x) => !refill.includes(x))]);
}
function drawChips() {
  $('chips').innerHTML = refill.length
    ? refill.map((s, i) => `<span class="chip">${i + 1}. ${esc(s)}<button title="移除" onclick="saveRefill(refill.filter(x=>x!=='${esc(s)}'))">✕</button></span>`).join('')
    : '<span class="sub">（空的：拍照順序只會列收購後台標記、還沒上架的新商品）</span>';
}
$('refillInput').addEventListener('keydown', (e) => { if (e.key === 'Enter') addRefill(); });
$('replyText').addEventListener('keydown', (e) => { if (e.key === 'Enter') send(e.target.value); });
$('serials').addEventListener('keydown', (e) => { if (e.key === 'Enter') runAddPhotos(); });

// ---------- 商品狀態 ----------
const STAGES = { new: ['🛒 還沒上架', 'b-new'], draft: ['📝 飛鼠草稿', 'b-draft'], shopee: ['🦐 蝦皮待補', 'b-shopee'], danger: ['🚨 要馬上處理', 'b-danger'], done: ['✅ 完成', 'b-done'], error: ['⚠️ 資料不全', 'b-error'] };
let items = [], filter = 'todo';
async function loadStatus() {
  $('statusInfo').textContent = '讀取中…（會到官網確認，約 10 秒）';
  const r = await (await fetch('/api/status')).json().catch(() => ({ ok: false, msg: '控制台沒有回應' }));
  if (!r.ok) { $('statusInfo').textContent = '讀取失敗'; $('table').innerHTML = `<div class="empty">${esc(r.msg)}</div>`; return; }
  items = r.items;
  $('inboxInfo').textContent = r.inbox ? `收件匣現在有 ${r.inbox} 張照片` : '收件匣是空的，照片全部放進來';
  $('statusInfo').textContent = `收購後台有標記的 ${items.length} 件`;
  drawTable();
}
function drawTable() {
  const count = (f) => items.filter(f).length;
  const F = { todo: ['待處理', (x) => x.stage !== 'done'], all: ['全部', () => true], new: ['還沒上架', (x) => x.stage === 'new'], draft: ['草稿', (x) => x.stage === 'draft'], shopee: ['蝦皮待補', (x) => x.stage === 'shopee' || x.stage === 'danger'], done: ['完成', (x) => x.stage === 'done'] };
  $('filters').innerHTML = Object.entries(F).map(([k, [t, f]]) => `<button class="${filter === k ? 'on' : ''}" onclick="filter='${k}';drawTable()">${t} ${count(f)}</button>`).join('');
  const list = items.filter(F[filter][1]);
  if (!list.length) { $('table').innerHTML = '<div class="empty">沒有商品 🎉</div>'; return; }
  $('table').innerHTML = `<table><tr><th>狀態</th><th>商品</th><th>照片</th><th></th></tr>${list.map((x) => {
    const [t, c] = STAGES[x.stage] || ['?', 'b-error'];
    const inList = refill.includes(x.serial);
    const live = x.stage === 'shopee' || x.stage === 'done' || x.stage === 'danger';
    return `<tr><td><span class="badge ${c}">${t}</span></td>
      <td><div class="name">${esc(x.name)}</div><div class="sub">序號 ${esc(x.serial)}　${x.type}　售價 ${esc(x.price || '-')}</div><div class="sub">${esc(x.hint)}</div></td>
      <td>${x.photos ? `${x.photos} 張${x.pending ? '<div class="sub">還沒傳</div>' : ''}` : '<span class="sub">沒有</span>'}</td>
      <td style="white-space:nowrap">${live ? `<button onclick="${inList ? `saveRefill(refill.filter(s=>s!=='${esc(x.serial)}'))` : `addRefill('${esc(x.serial)}')`}">${inList ? '✓ 在補照片清單' : '＋ 補照片清單'}</button>` : ''}
        ${x.photos && x.stage !== 'new' ? `<button onclick="run('addphotos',['${esc(x.serial)}'])">🖼️ 傳照片</button>` : ''}</td></tr>`;
  }).join('')}</table>`;
}
loadRefill(); loadStatus();
</script>
