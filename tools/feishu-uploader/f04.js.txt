// 自動更新：從收購後台網站下載最新版的上架程式，覆蓋舊檔
// 只會更新程式檔（.js、.bat、使用說明），不會動金鑰、紀錄 csv、照片、補照片清單、chrome-profile
// 用法：雙擊「更新程式.bat」（或 node update.js）
const fs = require('fs');
const path = require('path');
const crypto = require('crypto');
const dns = require('dns');
dns.setDefaultResultOrder && dns.setDefaultResultOrder('ipv4first');

const BASE = 'https://precious-otter-3873db.netlify.app/tools/feishu-uploader/';
const VERSION_FILE = path.join(__dirname, '版本.txt');
// 這些永遠不會被覆蓋或刪除
const PROTECTED = /(^|[\\/])(airtable-token\.txt|.*\.csv|補照片清單\.txt|飛鼠新增商品網址\.txt|平台分類選項\.txt|chrome-profile|node_modules|商品照片)([\\/]|$)/i;

async function get(url, tries = 4) {
  for (let i = 1; ; i++) {
    try {
      const res = await fetch(url + (url.includes('?') ? '&' : '?') + 't=' + Date.now(), { cache: 'no-store' });
      if (!res.ok) throw new Error('HTTP ' + res.status);
      return Buffer.from(await res.arrayBuffer());
    } catch (e) {
      if (i >= tries) throw new Error(`下載失敗：${url}（${e.message}）。請檢查網路再試一次`);
      await new Promise((r) => setTimeout(r, i * 2000));
    }
  }
}
const sha = (buf) => crypto.createHash('sha256').update(buf).digest('hex');

(async () => {
  console.log('檢查最新版本…');
  const manifest = JSON.parse((await get(BASE + 'manifest.json')).toString('utf8'));
  const local = fs.existsSync(VERSION_FILE) ? fs.readFileSync(VERSION_FILE, 'utf8').trim() : '（未知）';
  console.log(`目前版本：${local}　最新版本：${manifest.version}`);

  const stamp = new Date().toISOString().replace(/[-:T]/g, '').slice(0, 12);
  const backupDir = path.join(__dirname, '舊版備份', stamp);
  const backup = (p) => {
    const dest = path.join(backupDir, path.relative(__dirname, p));
    fs.mkdirSync(path.dirname(dest), { recursive: true });
    fs.renameSync(p, dest);
  };

  let changed = 0, pkgChanged = false;
  for (const f of manifest.files) {
    if (PROTECTED.test(f.path)) continue;
    const dest = path.join(__dirname, f.path);
    if (fs.existsSync(dest) && sha(fs.readFileSync(dest)) === f.sha256) continue;
    const buf = await get(BASE + f.url);
    if (sha(buf) !== f.sha256) throw new Error(`檔案下載不完整：${f.path}，請再試一次`);
    if (fs.existsSync(dest)) backup(dest);
    fs.mkdirSync(path.dirname(dest), { recursive: true });
    fs.writeFileSync(dest, buf);
    if (dest.endsWith('.command')) fs.chmodSync(dest, 0o755); // Mac 的按鈕要能雙擊執行
    console.log('  ✅ 更新 ' + f.path);
    changed++;
    if (f.path === 'package.json') pkgChanged = true;
  }
  // 舊版留下、已經不用的檔案：移到 舊版備份（不會直接刪掉）
  for (const old of manifest.remove || []) {
    const p = path.join(__dirname, old);
    if (!PROTECTED.test(old) && fs.existsSync(p)) { backup(p); console.log('  🗑️ 移走舊檔 ' + old); changed++; }
  }
  fs.writeFileSync(VERSION_FILE, manifest.version, 'utf8');
  if (process.platform !== 'win32') for (const f of fs.readdirSync(__dirname)) if (f.endsWith('.command')) fs.chmodSync(path.join(__dirname, f), 0o755);

  if (pkgChanged) {
    console.log('\n套件有更新，執行 npm install…');
    require('child_process').execSync('npm install', { cwd: __dirname, stdio: 'inherit' });
  }
  console.log(changed ? `\n完成！更新了 ${changed} 個檔案（舊檔備份在 舊版備份\\${stamp}）` : '\n已經是最新版，不用更新。');
  if (manifest.note) console.log('\n這次更新：' + manifest.note);
})().catch((e) => { console.error('\n❌ ' + (e.message || e)); process.exit(1); });
